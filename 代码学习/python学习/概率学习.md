### 实际使用
#### PCA
```python
# 1. 导入所需库
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler

# 2. 构建学生成绩数据集（和示例完全一致）
data = {
    'Nom': ['jean', 'alan', 'annie', 'monique', 'didier', 'andre', 'pierre', 'brigitte', 'evelyne'],
    'MATH': [6, 8, 6, 14.5, 14, 11, 5.5, 13, 9],
    'PHYS': [6, 8, 7, 14.5, 14, 10, 7, 12.5, 9.5],
    'FRAN': [5, 8, 11, 15.5, 12, 5.5, 14, 8.5, 12.5],
    'ANGL': [5.5, 8, 9.5, 15, 12.5, 7, 11.5, 9.5, 12]
}

# 转换为DataFrame
df = pd.DataFrame(data)

# 提取数值型变量（用于PCA分析，排除姓名列）
X = df[['MATH', 'PHYS', 'FRAN', 'ANGL']]
print("=== 原始数值数据 ===")
print(X)
print("\n")

# 3. 数据预处理（中心化，示例中未做标准化，保持和原文一致）
# 注：sklearn的PCA默认会中心化数据，无需手动处理；若需标准化，可打开下方注释
# scaler = StandardScaler()
# X_scaled = scaler.fit_transform(X)
X_processed = X.values  # 直接使用原始数据（和示例方差-协方差矩阵匹配）

# 4. 执行PCA（ACP）分析
pca = PCA()  # 初始化PCA，默认保留所有主成分
X_pca = pca.fit_transform(X_processed)  # 拟合并转换数据

# 5. 提取关键结果（和示例对应）
print("=== PCA关键结果 ===")
# 5.1 特征值（解释的方差）
eigenvalues = pca.explained_variance_
print("1. 特征值（各主成分方差）：")
print(np.round(eigenvalues, 4))  # 保留4位小数，和示例[31.7e+01, 13.5e+01, 3.67e-02, 1.19e-02]一致

# 5.2 方差解释率（单个主成分占总方差的比例）
explained_variance_ratio = pca.explained_variance_ratio_
print("2. 单个主成分方差解释率：")
print(np.round(explained_variance_ratio, 4))

# 5.3 累计方差解释率
cumulative_variance_ratio = np.cumsum(explained_variance_ratio)
print("3. 累计方差解释率：")
print(np.round(cumulative_variance_ratio, 4))

# 5.4 特征向量矩阵（按列排列，和示例一致）
eigenvectors = pca.components_.T  # 转置后，每一列是一个特征向量（主成分）
print("4. 特征向量矩阵（每列对应一个主成分）：")
print(np.round(eigenvectors, 4))
print("\n")

# 6. 验证总惯性（总方差）
total_variance = np.sum(eigenvalues)
print("=== 总惯性验证 ===")
print(f"总方差（总惯性）：{np.round(total_variance, 4)}")  # 应等于45.3472，和示例一致
print("\n")

# 7. 绘制帕累托图（特征值+累计解释率，示例要求手动绘制）
plt.rcParams['font.sans-serif'] = ['SimHei']  # 解决中文显示问题
plt.rcParams['axes.unicode_minus'] = False

# 创建画布和双坐标轴
fig, ax1 = plt.subplots(figsize=(10, 6))

# 坐标轴1：柱状图（单个特征值/方差解释率）
x = np.arange(1, len(eigenvalues)+1)  # 主成分序号（1,2,3,4）
ax1.bar(x, explained_variance_ratio, color='skyblue', alpha=0.7, label='单个方差解释率')
ax1.set_xlabel('主成分序号（VP1-VP4）')
ax1.set_ylabel('单个方差解释率', color='blue')
ax1.tick_params(axis='y', labelcolor='blue')
ax1.set_xticks(x)
ax1.set_xticklabels([f'VP{i}' for i in x])

# 坐标轴2：折线图（累计方差解释率）
ax2 = ax1.twinx()
ax2.plot(x, cumulative_variance_ratio, color='red', marker='o', linewidth=2, label='累计方差解释率')
ax2.set_ylabel('累计方差解释率', color='red')
ax2.tick_params(axis='y', labelcolor='red')
ax2.set_ylim(0, 1.1)  # 累计解释率范围0-1

# 添加数值标注（更清晰）
for i, (ratio, cum_ratio) in enumerate(zip(explained_variance_ratio, cumulative_variance_ratio)):
    ax1.text(i+1, ratio+0.02, f'{ratio:.4f}', ha='center', fontsize=9)
    ax2.text(i+1, cum_ratio+0.02, f'{cum_ratio:.4f}', ha='center', fontsize=9, color='red')

# 添加标题和网格
plt.title('PCA帕累托图（方差解释率）')
ax1.grid(axis='y', alpha=0.3)

# 合并图例
lines1, labels1 = ax1.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax1.legend(lines1 + lines2, labels1 + labels2, loc='lower right')

# 保存图片（可选，注释掉则直接显示）
# plt.savefig('pca_pareto.png', dpi=300, bbox_inches='tight')

# 显示图形
plt.show()

# 8. 绘制相关系数矩阵热力图（补充，和示例前期分析一致）
plt.figure(figsize=(8, 6))
corr_matrix = X.corr()
sns.heatmap(corr_matrix, annot=True, cmap='Greys', fmt='.6f', linewidths=0.5)
plt.title('相关系数矩阵热力图')
plt.show()
```

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn import datasets
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler

# 全局配置
plt.rcParams['font.sans-serif'] = ['SimHei']
plt.rcParams['axes.unicode_minus'] = False
sns.set_style("whitegrid")

# 1. 加载数据
iris = datasets.load_iris()

class_names = iris.target_names
df_iris = pd.DataFrame(X, columns=feature_names)
df_iris['类别'] = [class_names[i] for i in y]

# 2. 数据标准化
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# 3. PCA 计算
pca = PCA()
X_pca = pca.fit_transform(X_scaled)
pca_eigenvalues = pca.explained_variance_
pca_explained_ratio = pca.explained_variance_ratio_
pca_cumulative_ratio = np.cumsum(pca_explained_ratio)

# 4. 输出结果
print("=== PCA 结果 ===")
print(f"特征值：{np.round(pca_eigenvalues, 4)}")
print(f"方差解释率：{np.round(pca_explained_ratio, 4)}")

# 5. 可视化
# 帕累托图
fig, ax1 = plt.subplots(figsize=(10, 6))
x = np.arange(1, len(pca_eigenvalues)+1)
ax1.bar(x, pca_explained_ratio, color='skyblue', alpha=0.7)
ax2 = ax1.twinx()

plt.title('PCA 帕累托图（方差解释率）')
plt.show()

# 投影散点图
plt.figure(figsize=(8, 6))
for idx, cls in enumerate(class_names):
    plt.scatter(X_pca[y==idx, 0], X_pca[y==idx, 1], label=cls, alpha=0.8)
plt.xlabel(f'PC1 ({pca_explained_ratio[0]:.2%})')
plt.ylabel(f'PC2 ({pca_explained_ratio[1]:.2%})')
plt.title('PCA 投影散点图（PC1 vs PC2）')
plt.legend()
plt.show()

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn import datasets
from sklearn.discriminant_analysis import LinearDiscriminantAnalysis
from sklearn.preprocessing import StandardScaler

# 全局配置
plt.rcParams['font.sans-serif'] = ['SimHei']
plt.rcParams['axes.unicode_minus'] = False
sns.set_style("whitegrid")

# 1. 加载数据
iris = datasets.load_iris()
X = iris.data
y = iris.target
feature_names = iris.feature_names
class_names = iris.target_names
df_iris = pd.DataFrame(X, columns=feature_names)
df_iris['类别'] = [class_names[i] for i in y]

# 2. 数据标准化
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# 3. LDA 计算
lda = LinearDiscriminantAnalysis()
X_lda = lda.fit_transform(X_scaled, y)
lda_eigenvalues = lda.explained_variance_
lda_explained_ratio = lda.explained_variance_ratio_
lda_cumulative_ratio = np.cumsum(lda_explained_ratio)

print("=== LDA 结果 ===")
print(f"特征值（判别能力）：{np.round(lda_eigenvalues, 4)}")
print(f"判别能力解释率：{np.round(lda_explained_ratio, 4)}")
print(f"累计解释率：{np.round(lda_cumulative_ratio, 4)}")

# 5. 可视化
# 投影散点图
plt.figure(figsize=(8, 6))
for idx, cls in enumerate(class_names):
    plt.scatter(X_lda[y==idx, 0], X_lda[y==idx, 1], label=cls, alpha=0.8, s=60)
plt.xlabel(f'LD1 ({lda_explained_ratio[0]:.2%})')
plt.ylabel(f'LD2 ({lda_explained_ratio[1]:.2%})')
plt.title('LDA 投影散点图（LD1 vs LD2）')
plt.legend()
plt.show()

# 6. 分类预测演示
sample = np.array([[5.1, 3.5, 1.4, 0.2]])  # 模拟Setosa样本
sample_scaled = scaler.transform(sample)
pred_label = lda.predict(sample_scaled)
pred_proba = lda.predict_proba(sample_scaled)
print("\n=== 预测结果 ===")
print(f"预测类别：{class_names[pred_label[0]]}")
print(f"类别概率：{dict(zip(class_names, np.round(pred_proba[0], 4)))}")

### 代码学习
#### 纯 PCA 代码关键语句解释
##### 数据标准化相关
```python
scaler = StandardScaler() 
X_scaled = scaler.fit_transform(X)
```
- 对原始特征数据进行「标准化」（均值为 0，方差为 1）
- `StandardScaler()`：初始化标准化器，默认实现「(x - 均值) / 标准差」的转换。
- `fit_transform()`：一次性完成「计算均值 / 标准差（fit）」和「应用转换（transform）」，返回标准化后的特征矩阵 `X_scaled`
##### PCA 核心计算
```python
pca = PCA() 
X_pca = pca.fit_transform(X_scaled)
```
- 初始化 PCA 模型，并用标准化数据拟合模型，同时完成数据降维转换
- `PCA()`：初始化 PCA 模型，默认参数 `n_components=None`，表示保留所有主成分（即主成分数 = 原始特征数，这里是 4 个），后续可通过方差解释率选择保留的数量。
- `fit`：分析标准化数据，计算协方差矩阵、特征值、特征向量（主成分轴）
- `transform`：将原始高维数据（4 维）投影到计算出的主成分轴上，得到降维后的新数据 `X_pca`（形状为 [150, 4]，150 个样本，4 个主成分）
##### 提取 PCA 核心结果
```python
pca_explained_ratio = pca.explained_variance_ratio_
```
- 提取「每个主成分的方差解释率」，即每个主成分能保留原始数据总方差的比例
##### PCA 可视化核心
```python
plt.scatter(X_pca[y==idx, 0], X_pca[y==idx, 1], label=cls, alpha=0.8)
```
- 绘制前两个主成分（PC1 vs PC2）的投影散点图，可视化降维后的数据分布。
- `X_pca[y==idx, 0]`：筛选出「类别为 `idx`」的所有样本，取它们的「第一主成分（PC1）」数据作为横坐标。
- `X_pca[y==idx, 1]`：同理，取「第二主成分（PC2）」数据作为纵坐标。
- 这里用到 `y`（类别标签）只是为了给不同类别上色，方便观察分布，PCA 本身的计算过程不需要 `y`（这是无监督的核心体现）。
#### 纯 LDA 代码关键语句解释
##### LDA 核心计算
```python
lda = LinearDiscriminantAnalysis() 
X_lda = lda.fit_transform(X_scaled, y)
```
- `fit_transform()` 传入了两个参数 `(X_scaled, y)`，`y` 是类别标签，LDA 必须依赖标签才能计算「组间离散度矩阵 B」和「组内离散度矩阵 W」
- `LinearDiscriminantAnalysis()`：初始化 LDA 模型，默认保留所有有效判别轴（最多为「类别数 - 1」，这里 3 个类别，故保留 2 个判别轴）
- `fit`：分析数据，计算各组重心、组内离散度矩阵 W、组间离散度矩阵 B，求解 `W^{-1}B` 的特征值和特征向量（判别轴）
- `transform`：将原始高维数据投影到判别轴上，得到降维后的新数据 `X_lda`（形状为 [150, 2]，150 个样本，2 个判别轴）
##### 提取 LDA 核心结果
```python
pred_label = lda.predict(sample_scaled)
```
